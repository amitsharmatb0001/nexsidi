import axios from 'axios';

const apiClient = axios.create({
    baseURL: 'http://127.0.0.1:8000/api',
    withCredentials: true,
    headers: {
        'Accept': 'application/json',
        'Content-Type': 'application/json',
    }
});

apiClient.interceptors.request.use(config => {
    const token = localStorage.getItem('authToken');
    if (token) {
        config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
}, error => {
    return Promise.reject(error);
});

export default apiClient;

const ViewCartBar = ({ cart }) => {
    const navigate = useNavigate();
    
    // This check prevents the crash. If cart is null or empty, render nothing.
    if (!cart || !cart.items || cart.items.length === 0) {
        return null;
    }

    const itemCount = cart.items.reduce((sum, item) => sum + item.quantity, 0);
    const subtotal = cart.items.reduce((sum, item) => sum + (item.menu_item.price * item.quantity), 0);

    return (
        <div 
            onClick={() => navigate('/cart')}

            <div className="font-bold">
                {itemCount} {itemCount > 1 ? 'items' : 'item'} | ₹{subtotal}
            </div>
            <div className="font-bold flex items-center">
                View Cart 
                <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5 ml-2" fill="none" viewBox="0 0 24 24" stroke="currentColor">

};

import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import apiClient from '../api/apiClient';

const CartPage = ({ cart, onUpdateCart, onPlaceOrder }) => {
    const navigate = useNavigate();
    const [addresses, setAddresses] = useState([]);
    const [selectedAddressId, setSelectedAddressId] = useState('');

    useEffect(() => {
        const fetchAddresses = async () => {
            try {
                const response = await apiClient.get('/addresses');
                setAddresses(response.data);
                if (response.data.length > 0) {
                    setSelectedAddressId(response.data[0].id);
                }
            } catch (error) {
                console.error("Failed to fetch addresses for cart:", error);
            }
        };
        fetchAddresses();
    }, []);
    
    const cartItems = cart?.items || [];
    const subtotal = cartItems.reduce((sum, item) => sum + (item.menu_item.price * item.quantity), 0);
    const deliveryFee = 30;
    const taxes = subtotal * 0.05;
    const total = subtotal + deliveryFee + taxes;

    const handleProceedToCheckout = () => {
        if (!selectedAddressId) {
            alert("Please select a delivery address."); // This can be a toast notification
            return;
        }
        onPlaceOrder(selectedAddressId);
    };

    return (
        <div className="container mx-auto px-4 py-8">
            <h2 className="text-3xl font-bold mb-6">My Cart</h2>
            {cartItems.length === 0 ? (
                <div className="text-center py-12 bg-white rounded-lg shadow">
                    <p className="text-xl text-gray-600">Your cart is empty.</p>
                    <button onClick={() => navigate('/')} className="mt-4 bg-yellow-500 text-white font-bold py-2 px-6 rounded hover:bg-yellow-600">Start Ordering</button>
                </div>
            ) : (
                <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
                    <div className="lg:col-span-2 space-y-6">
                        <div className="bg-white rounded-lg shadow p-6 space-y-4">
                            {cartItems.map(item => (
                                <div key={item.id} className="flex items-center justify-between border-b pb-4 last:border-b-0">
                                    <div>
                                        <p className="font-bold">{item.menu_item.name}</p>
                                        <p className="text-gray-600">₹{item.menu_item.price}</p>
                                    </div>
                                    <div className="flex items-center">
                                        <button onClick={() => onUpdateCart(item.id, item.quantity - 1)} className="bg-gray-200 px-3 py-1 rounded-l">-</button>
                                        <span className="px-4">{item.quantity}</span>
                                        <button onClick={() => onUpdateCart(item.id, item.quantity + 1)} className="bg-gray-200 px-3 py-1 rounded-r">+</button>
                                    </div>
                                    <p className="font-bold">₹{item.menu_item.price * item.quantity}</p>
                                </div>
                            ))}
                        </div>
                        <div className="bg-white rounded-lg shadow p-6">
                            <h4 className="text-xl font-bold mb-4">Select Delivery Address</h4>
                            <div className="space-y-3">
                                {addresses.map(addr => (
                                    <label key={addr.id} className="flex items-center p-3 border rounded-lg cursor-pointer">
                                        <input type="radio" name="address" value={addr.id} checked={selectedAddressId === addr.id} onChange={(e) => setSelectedAddressId(Number(e.target.value))} />
                                        <div className="ml-3"><p className="font-semibold">{addr.type}</p><p className="text-gray-600 text-sm">{addr.address}</p></div>
                                    </label>
                                ))}
                                {addresses.length === 0 && <p className="text-gray-500 text-sm">Please add an address on your profile page.</p>}
                            </div>
                        </div>
                    </div>
                    <div className="lg:col-span-1 bg-white rounded-lg shadow p-6 h-fit">
                        <h3 className="text-xl font-bold mb-4">Order Summary</h3>
                        <div className="space-y-2 text-gray-700">
                           <div className="flex justify-between"><span>Subtotal</span><span>₹{subtotal.toFixed(2)}</span></div>
                            <div className="flex justify-between"><span>Delivery Fee</span><span>₹{deliveryFee.toFixed(2)}</span></div>
                            <div className="flex justify-between"><span>Taxes & Charges</span><span>₹{taxes.toFixed(2)}</span></div>
                            <div className="border-t my-2"></div>
                            <div className="flex justify-between font-bold text-lg"><span>Total</span><span>₹{total.toFixed(2)}</span></div>
                        </div>
                        <button onClick={handleProceedToCheckout} disabled={!selectedAddressId} className="mt-6 w-full bg-green-600 text-white font-bold py-3 rounded hover:bg-green-700 disabled:bg-gray-400">
                            Place Order
                        </button>
                    </div>
                </div>
            )}
        </div>
    );
};

export default CartPage;

import React, { useState, useEffect, useCallback } from 'react';
import { useNavigate } from 'react-router-dom';
import { SearchIcon } from '../components/Icons';
import RestaurantCard from '../restaurants/components/RestaurantCard';
import { moods } from '../data/mockData';
import apiClient from '../api/apiClient';
import ViewCartBar from '../cart/components/ViewCartBar';
import debounce from 'lodash/debounce'; // ✅ still keeping debounce fix

const HomePage = ({ cart }) => {
  const [allRestaurants, setAllRestaurants] = useState([]);
  const [filteredRestaurants, setFilteredRestaurants] = useState([]);
  const [loading, setLoading] = useState(true);
  const [searchTerm, setSearchTerm] = useState('');
  const [filters, setFilters] = useState({ rating: null, veg_only: false });
  const navigate = useNavigate();

  // Fetch ALL restaurants just once on initial load
  useEffect(() => {
    const fetchInitialRestaurants = async () => {
      setLoading(true);
      try {
        const response = await apiClient.get('/restaurants/search'); 
        setAllRestaurants(response.data);
        setFilteredRestaurants(response.data);
      } catch (error) {
        console.error("Failed to fetch initial restaurants:", error);
      } finally {
        setLoading(false);
      }
    };
    fetchInitialRestaurants();
  }, []);

  const applyFilters = useCallback((search, currentFilters) => {
    let results = [...allRestaurants];

    if (search) {
      const lowercasedSearch = search.toLowerCase();
      results = results.filter(restaurant => 
        restaurant.name.toLowerCase().includes(lowercasedSearch) ||
        (restaurant.menu_items && restaurant.menu_items.some(item => item.name.toLowerCase().includes(lowercasedSearch)))
      );
    }

    if (currentFilters.veg_only) {
      results = results.filter(restaurant => restaurant.is_veg);
    }

    if (currentFilters.rating) {
      results = results.filter(restaurant => restaurant.rating >= currentFilters.rating);
    }

    setFilteredRestaurants(results);
  }, [allRestaurants]);

  // ✅ apply debounce to filter handler
  const debouncedApplyFilters = useCallback(
    debounce(applyFilters, 300),
    [applyFilters]
  );

  useEffect(() => {
    debouncedApplyFilters(searchTerm, filters);
  }, [searchTerm, filters, debouncedApplyFilters]);

  const handleFilterChange = (filterName, value) => {
    setFilters(prev => ({ ...prev, [filterName]: prev[filterName] === value ? null : value }));
  };

  const handleVegToggle = () => {
    setFilters(prev => ({ ...prev, veg_only: !prev.veg_only }));
  };

  const handleMoodClick = (moodSearchTerm) => {
    setSearchTerm(moodSearchTerm);
  };

  const hasSearchTerm = searchTerm.length > 0;

  return (
    <div className="container mx-auto px-4 py-8 pb-24">
      {/* Heading & Search */}
      <div className="text-center mb-12">
        <h2 className="text-4xl font-extrabold text-text-primary">What are you craving today?</h2>
        <div className="mt-6 max-w-lg mx-auto">
          <div className="relative flex items-center">
            <input 
              type="text" 
              placeholder="Search for restaurants or dishes..." 
              className="w-full bg-surface p-4 rounded-full shadow-lg focus:outline-none focus:ring-2 focus:ring-primary transition-shadow text-text-primary"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
            />
            <button className="absolute right-3 bg-primary text-white rounded-full p-2 hover:opacity-90">
              <SearchIcon />
            </button>
          </div>
        </div>
      </div>

      {/* Search results */}
      {hasSearchTerm && (
        <div className="mb-12">
          <h3 className="text-2xl font-bold text-text-primary mb-4">Search Results for "{searchTerm}"</h3>
          {filteredRestaurants.length > 0 ? (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
              {filteredRestaurants.map(r => <RestaurantCard key={r.id} restaurant={r} />)}
            </div>
          ) : (
            <p className="text-text-secondary text-center py-10">No results found.</p>
          )}
        </div>
      )}

      {/* Filters, Moods & Popular Restaurants */}

import React, { useState, useEffect } from 'react';
import { Routes, Route, useNavigate, useLocation } from 'react-router-dom';
import apiClient from './api/apiClient';

// Components & Pages
import Header from './components/Header';
import Footer from './components/Footer';
import HomePage from './home/HomePage';
import LoginPage from './authentication/LoginPage';
import RegisterPage from './authentication/RegisterPage';
import RestaurantDetailPage from './restaurants/RestaurantDetailPage';
import CartPage from './cart/CartPage';
import ProfilePage from './profile/ProfilePage';
import OrderSuccessPage from './orders/OrderSuccessPage';

// Notifications
import { ToastContainer } from 'react-toastify';
import 'react-toastify/dist/ReactToastify.css';
import { notify } from './utils/notifications';

export default function App() {
    const [cart, setCart] = useState(null);
    const [user, setUser] = useState(null);
    const navigate = useNavigate();
    const location = useLocation();

    // =================================================================
    // THIS IS THE FIX: This useEffect runs once on app load.
    // It checks localStorage for a saved session and restores it.
    // =================================================================
    useEffect(() => {
        const bootstrapApp = async () => {
            const storedToken = localStorage.getItem('authToken');
            const storedUser = localStorage.getItem('authUser');
            
            if (storedToken && storedUser) {
                // This line is crucial: It sets the auth header for all future API calls
                apiClient.defaults.headers.common['Authorization'] = `Bearer ${storedToken}`;
                
                setUser(JSON.parse(storedUser));
                try {
                    // We still fetch the cart to get the latest data
                    const response = await apiClient.get('/cart');
                    setCart(response.data);
                } catch (error) {
                    console.error("Could not fetch cart on initial load:", error);
                    // If the token is expired/invalid, the API will likely return a 401
                    if (error.response && error.response.status === 401) {
                        handleLogout(); // Log the user out gracefully
                    }
                }
            }
        };
        bootstrapApp();
    }, []); // The empty dependency array [] ensures this runs only once.

    // --- Auth & Cart Handlers ---
    const handleLoginSuccess = async (userData, authToken) => {
        localStorage.setItem('authUser', JSON.stringify(userData));
        localStorage.setItem('authToken', authToken);
        // This line was missing in your original login handler
        apiClient.defaults.headers.common['Authorization'] = `Bearer ${authToken}`;
        setUser(userData);
        try {
            const response = await apiClient.get('/cart');
            setCart(response.data);
        } catch (error) {
            console.error("Could not fetch cart after login:", error);
        }
        navigate('/');
    };

    const handleLogout = () => {
        setUser(null);
        setCart(null);
        localStorage.removeItem('authUser');
        localStorage.removeItem('authToken');
        delete apiClient.defaults.headers.common['Authorization'];
        navigate('/');
    };
    
    const handleAddToCart = async (menuItemId, quantity) => {
        try {
            const response = await apiClient.post('/cart/add', { menu_item_id: menuItemId, quantity });
            setCart(response.data.cart);
            notify.success("Item added to cart!");
        } catch (error) {
            console.error("Failed to add item to cart:", error);
            notify.error(error.response?.data?.message || "Could not add item.");
        }
    };
    
    const handleUpdateCart = async (cartItemId, newQuantity) => {
        try {
            if (newQuantity <= 0) {
                await apiClient.delete(`/cart/remove/${cartItemId}`);
            } else {
                await apiClient.put(`/cart/update/${cartItemId}`, { quantity: newQuantity });
            }
            const response = await apiClient.get('/cart');
            setCart(response.data);
        } catch (error) {
            console.error("Failed to update cart:", error);
            notify.error("Failed to update cart.");
        }
    };

    const handlePlaceOrder = async (userAddressId) => {
        try {
            await apiClient.post('/orders', { user_address_id: userAddressId });
            setCart(null);
            navigate('/order-success');
        } catch (error) {
            console.error("Failed to place order:", error);
            notify.error(error.response?.data?.message || "Could not place order.");
        }
    };

    // --- Computed values for header ---
    const cartCount = cart?.items?.reduce((sum, item) => sum + item.quantity, 0) || 0;
    const hideHeaderCart = location.pathname.startsWith('/restaurants/') || location.pathname === '/';

    return (
        <div className="bg-background text-text-primary min-h-screen font-sans">
            <ToastContainer />
            <Header cartCount={cartCount} isLoggedIn={!!user} hideCart={hideHeaderCart} />
            <main>
                <Routes>
                    <Route path="/" element={<HomePage cart={cart} />} />
                    <Route 
                        path="/restaurants/:id" 
                        element={
                            <RestaurantDetailPage 
                                cart={cart}
                                onAddToCart={handleAddToCart} 
                                onUpdateCart={handleUpdateCart}
                                user={user} 
                            />
                        } 
                    />
                    <Route 
                        path="/cart" 
                        element={
                            user 
                            ? <CartPage 
                                cart={cart} 
                                onUpdateCart={handleUpdateCart} 
                                onPlaceOrder={handlePlaceOrder} 
                            /> 
                            : <LoginPage onLoginSuccess={handleLoginSuccess} />
                        } 
                    />
                    <Route 
                        path="/profile" 
                        element={
                            user 
                            ? <ProfilePage 
                                user={user} 
                                onLogout={handleLogout} 
                            /> 
                            : <LoginPage onLoginSuccess={handleLoginSuccess} />
                        } 
                    />
                    <Route path="/login" element={<LoginPage onLoginSuccess={handleLoginSuccess} />} />
                    <Route path="/register" element={<RegisterPage />} />
                    <Route path="/order-success" element={<OrderSuccessPage />} />
                </Routes>
            </main>
            <Footer />
        </div>
    );
}
